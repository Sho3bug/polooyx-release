cmake_minimum_required(VERSION 3.22)

project(POLOOYX_RELEASE VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")   # static CRT: no VC++ redist needed
endif()

option(PRL_BUILD_TESTS "Build the offline DSP test harness" ON)

# JUCE: use a local checkout if present, otherwise download JUCE 8.0.4 at configure time
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/JUCE/CMakeLists.txt")
    add_subdirectory(JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.4
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

# Inter typeface (SIL OFL) — generated by tools/make_fonts.py (CI does this). Without it the UI uses the system sans-serif.
set(PRL_HAS_INTER 0)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Inter-Regular.ttf")
    set(PRL_HAS_INTER 1)
    juce_add_binary_data(PrlFonts
        HEADER_NAME PrlFonts.h
        NAMESPACE PrlFonts
        SOURCES
            Resources/Inter-Regular.ttf
            Resources/Inter-Medium.ttf
            Resources/Inter-SemiBold.ttf)
endif()

juce_add_plugin(PolooyxRelease
    COMPANY_NAME "Polooyx"
    COMPANY_WEBSITE ""
    BUNDLE_ID "com.polooyx.releaseengine"
    PLUGIN_MANUFACTURER_CODE Plxy
    PLUGIN_CODE Plr1
    PRODUCT_NAME "POLOOYX RELEASE ENGINE"
    DESCRIPTION "POLOOYX Release Engine - stereo mix analysis and mastering"
    VERSION 1.0.0
    FORMATS VST3 Standalone
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Fx Mastering Analyzer
    VST3_AUTO_MANIFEST FALSE)

set(PRL_SOURCES
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/DSP/MasterChain.cpp
    Source/Analysis/MixDiagnostics.cpp
    Source/Intelligence/AutoMasterEngine.cpp)

target_sources(PolooyxRelease PRIVATE ${PRL_SOURCES})
target_include_directories(PolooyxRelease PRIVATE Source)

set(PRL_DEFS
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0
    PRL_HAS_INTER=${PRL_HAS_INTER})

target_compile_definitions(PolooyxRelease PUBLIC ${PRL_DEFS})

target_link_libraries(PolooyxRelease
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

if(PRL_HAS_INTER)
    target_link_libraries(PolooyxRelease PRIVATE PrlFonts)
endif()

if(PRL_BUILD_TESTS)
    juce_add_console_app(PolooyxReleaseTests PRODUCT_NAME "PolooyxReleaseTests")
    target_sources(PolooyxReleaseTests PRIVATE Tests/TestMain.cpp ${PRL_SOURCES})
    target_include_directories(PolooyxReleaseTests PRIVATE Source)
    target_compile_definitions(PolooyxReleaseTests PRIVATE ${PRL_DEFS} JUCE_MODAL_LOOPS_PERMITTED=1)
    target_link_libraries(PolooyxReleaseTests
        PRIVATE
            juce::juce_audio_utils
            juce::juce_dsp
            juce::juce_recommended_config_flags)
    if(PRL_HAS_INTER)
        target_link_libraries(PolooyxReleaseTests PRIVATE PrlFonts)
    endif()
    if(MSVC)
        # the harness keeps many processor instances on main()'s stack; Windows defaults to 1 MB
        target_link_options(PolooyxReleaseTests PRIVATE /STACK:33554432)
    endif()
endif()
